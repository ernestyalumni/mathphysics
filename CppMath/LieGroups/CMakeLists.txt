cmake_minimum_required(VERSION 3.20)
project(MathphysicsLieGroups LANGUAGES CXX)
option(LIE_GROUPS_SANITIZE "Address/undefined sanitizer checks" OFF)
add_library(lie_groups INTERFACE)
target_include_directories(lie_groups INTERFACE include)
target_compile_features(lie_groups INTERFACE cxx_std_20)
enable_testing()
add_executable(lie_groups_tests tests/test_lie_groups.cpp)
target_link_libraries(lie_groups_tests PRIVATE lie_groups)
target_compile_options(lie_groups_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(LIE_GROUPS_SANITIZE)
  target_compile_options(lie_groups_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(lie_groups_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME lie_groups_contracts COMMAND lie_groups_tests)
add_executable(epsilon_regression tests/epsilon_regression.cpp)
target_link_libraries(epsilon_regression PRIVATE lie_groups)
add_test(NAME epsilon_regression COMMAND epsilon_regression)
add_executable(lie_groups_interchange tests/interchange.cpp)
target_link_libraries(lie_groups_interchange PRIVATE lie_groups)
target_compile_options(lie_groups_interchange PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(goldstein_rotation_tests tests/test_rotations.cpp)
target_link_libraries(goldstein_rotation_tests PRIVATE lie_groups)
target_compile_options(goldstein_rotation_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(LIE_GROUPS_SANITIZE)
  target_compile_options(goldstein_rotation_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(goldstein_rotation_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME goldstein_rotation_contracts COMMAND goldstein_rotation_tests)
add_executable(goldstein_rotation_interchange tests/rotation_interchange.cpp)
target_link_libraries(goldstein_rotation_interchange PRIVATE lie_groups)
target_compile_options(goldstein_rotation_interchange PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(rigid_body_tests tests/test_rigid_body.cpp)
target_link_libraries(rigid_body_tests PRIVATE lie_groups)
target_compile_options(rigid_body_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(LIE_GROUPS_SANITIZE)
  target_compile_options(rigid_body_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rigid_body_tests PRIVATE -fsanitize=address,undefined)
