\part{Quaternions, \texorpdfstring{$\mathrm{Spin}(3)$}{Spin(3)}, and \texorpdfstring{$SU(2)$}{SU(2)}}
\section{Conventions}
Use the Hamiltonian quaternions $\mathbb H$ with the multiplication rules
$i^2=j^2=k^2=-1$, $ij=-ji=k$, $jk=-kj=i$, $ki=-ik=j$, as defined in
the ring examples of the algebra part. Write the imaginary units as
\begin{equation}\label{quat:basis}
 (e_1,e_2,e_3)=(i,j,k),\qquad e_1e_2=e_3,\quad e_2e_3=e_1,\quad e_3e_1=e_2,
 \qquad e_a^2=-1 .
\end{equation}
A quaternion is written scalar first,
\begin{equation}\label{quat:scalar-first}
 q=q_0+q^ae_a,\qquad q_0\in\mathbb R,\quad (q^1,q^2,q^3)\in\mathbb R^3 .
\end{equation}
Its conjugate is $\bar q=q_0-q^ae_a$ and its norm satisfies
$|q|^2=q\bar q=q_0^2+q^aq^a$. A vector $\mathbf x\in\mathbb R^3$ is
identified with the pure quaternion $x^ae_a$.

For attitude, the unit quaternion $q$ carries body components to inertial
components by
\begin{equation}\label{quat:body-to-inertial}
 x_I^ae_a=q\,(x_B^ae_a)\,\bar q ,
\end{equation}
so it represents the same rotation $R$ as in Eq.~\eqref{gnc:attitude-kinematics},
where $\mathbf x_I=R\mathbf x_B$. The rotation acts on vectors (active form).

\subsection{Notation of these notes and of the sources}\label{quat:sec:notation}
Each book on this subject has its own symbols for the same objects. These
notes fix one set, which we call \emph{our} notation, and translate every
other author into it at the first use. Our notation is the one already used
in the Lie-groups part (\S\ref{dgdt:sec:lie-groups}), which follows
Kosmann-Schwarzbach \cite{dgdt:YKosmann-Schwarzbach2010}:
\begin{itemize}
 \item the Lie algebra of a matrix group $G$ is
  $\mathfrak g=\{\gamma'(0):\gamma \text{ a } C^1 \text{ curve in } G,\
  \gamma(0)=1\}$, and $X\in\mathfrak g$ if and only if $\exp(tX)\in G$ for
  all $t$ (Theorem~\ref{dgdt:Thm:5.1YKos}, which is Kosmann-Schwarzbach's
  Theorem~5.1);
 \item $\mathfrak{su}(2)=\{X\in\mathfrak{gl}(2,\mathbb C):X+X^\dagger=0,\
  \operatorname{tr}X=0\}$ with the basis $\xi_1,\xi_2,\xi_3$
  (\S\ref{dgdt:sec:KS-su2-bases});
 \item conjugation $\mathcal C_g(h)=ghg^{-1}$, the adjoint action
  $\mathrm{Ad}_g=D_1\mathcal C_g$, so that $\mathrm{Ad}_gY=gYg^{-1}$, and its
  derivative $\mathrm{ad}_XY=[X,Y]$
  (\S\ref{dgdt:sec:adjoint-representation}).
\end{itemize}
To these we add the symbols introduced in this part: the hat map
$\widehat{\boldsymbol\omega}=\omega^jE_j$ (Eq.~\eqref{quat:hat}), the map
$\Sigma(\mathbf x)=x^k\sigma_k$ to traceless Hermitian matrices
(Eq.~\eqref{quat:Sigma}), and the homomorphism
$\Phi\colon SU(2)\to SO(3)$ (Eq.~\eqref{quat:Phi}). The table lists

\section{The Lie algebra \texorpdfstring{$\mathfrak{so}(n)$}{so(n)} and Frankel's basis of \texorpdfstring{$\mathfrak{so}(3)$}{so(3)}}\label{quat:sec:so-n}
Let $G\subseteq GL(n,\mathbb R)$ be a closed matrix group. Its Lie algebra is
\begin{equation}\label{quat:matrix-lie-algebra}
 \mathfrak g=\{X\in\mathfrak{gl}(n,\mathbb R)\;:\;\exp(tX)\in G
 \text{ for all } t\in\mathbb R\},
\end{equation}
the characterization recorded from Kosmann-Schwarzbach in the Lie-groups
part. If $g(t)$ is a differentiable curve in $G$ with $g(0)=I$, then
$\frac{dg}{dt}(0)\in\mathfrak g$; this is how Frankel computes the basis
below.

\subsection{Orthogonal groups as regular level sets}\label{quat:sec:orthogonal-submanifold}
Frankel \cite{TFrankel2012}, Section~1.1d, pp.~9--11, proves that the
orthogonal matrices form a submanifold. We give the argument for $n\times n$
matrices, using the differential notation of our manifold notes. His
matrix $x$ becomes $A$, an orthogonal base point $x_0$ becomes $R$, and
$F_*v=w$ becomes $DF_R(H)=S$. Here $\operatorname{Sym}(n)$ denotes
the symmetric $n\times n$ matrices; its argument is the matrix size.
For $n=3$ this is Frankel's six-dimensional space $\mathrm{Sym}_6$.

Identify $\operatorname{Mat}(n,n)$ with $\mathbb R^{n^2}$ by its entries.
A symmetric matrix is specified by its diagonal and entries above it, so
\begin{equation}\label{quat:symmetric-dimension}
 \operatorname{Sym}(n)=\{S\in\operatorname{Mat}(n,n):S^T=S\},
 \qquad \dim\operatorname{Sym}(n)=\frac{n(n+1)}2.
\end{equation}
Define the smooth polynomial map
\begin{equation}\label{quat:orthogonality-map}
 F:\operatorname{Mat}(n,n)\longrightarrow\operatorname{Sym}(n),
 \qquad F(A)=A^TA-I.
\end{equation}
Its zero set is $O(n)$. For $n=3$ its entries are the column inner products
$(F(A))_{ij}=\sum_{k=1}^3 A_{ki}A_{kj}-\delta_{ij}$.
Since $(F(A))_{ij}=(F(A))_{ji}$, there are six distinct scalar equations.
To subtract six from nine, we must also prove independence of their
differentials at every orthogonal matrix.

Hold $A$ and the ambient direction $H\in\operatorname{Mat}(n,n)$ fixed.
Expanding before taking the derivative gives
\begin{align}
 F(A+sH)-F(A)
 &=s(H^TA+A^TH)+s^2H^TH,\notag\\
 DF_A(H)
 &=\lim_{s\to0}\frac{F(A+sH)-F(A)}s
   =H^TA+A^TH.\label{quat:orthogonality-differential}
\end{align}
Now let $R\in O(n)$ and let $S\in\operatorname{Sym}(n)$ be arbitrary.
Choose $H=\tfrac12RS$. Substitute $R^TR=I$ and $S^T=S$ into
Eq.~\eqref{quat:orthogonality-differential}:
\begin{align}
 DF_R\left(\tfrac12RS\right)
 &=\tfrac12S^TR^TR+\tfrac12R^TRS\notag\\
 &=\tfrac12S+\tfrac12S=S.\label{quat:orthogonality-right-inverse}
\end{align}
Thus $S\mapsto\tfrac12RS$ is a right inverse of $DF_R$, so $DF_R$
is surjective at every $R\in F^{-1}(0)$. Zero is a regular value.
The preimage theorem in the manifold part, equivalently Lee's regular
level set theorem \cite{dgdt:JLee2012}, Chapter~5, makes $O(n)$ an embedded
submanifold with
\begin{equation}\label{quat:dim-On}
 \dim O(n)=n^2-\frac{n(n+1)}2=\frac{n(n-1)}2.
\end{equation}
This proves differential independence on the level set, rather than
inferring it merely from the number of displayed equations.

For $R\in O(n)$, $(\det R)^2=\det(R^TR)=1$, so $\det R\in\{+1,-1\}$.
Moreover,
\[
 SO(n)=O(n)\cap\det^{-1}((0,\infty)).
\]
The determinant is continuous, so $SO(n)$ is open in $O(n)$. It inherits
the same dimension and local submanifold charts. Consequently $SO(3)$ is
an embedded three-dimensional submanifold of $\mathbb R^9$; the determinant
condition selects a sign and imposes no additional local scalar constraint.
Locally, the implicit function theorem lets us solve six matrix entries
smoothly in terms of three others, with the choice depending on the point.

The tangent space is the kernel of the differential:
\begin{align}
 T_RO(n)
 &=\{H:H^TR+R^TH=0\}\notag\\
 &=\{RX:X^T=-X\}.\label{quat:orthogonal-tangent}
\end{align}
Indeed, set $X=R^TH$, so $H=RX$ and $X^T+X=H^TR+R^TH$.
At $R\in SO(n)$ the tangent space is the same, because $SO(n)$ is open
in $O(n)$. At $R=I$ it consists of the skew-symmetric matrices.
Along a curve $R(t)\in SO(3)$ this gives $R^T\dot R\in\mathfrak{so}(3)$,
the angular-velocity construction in Eq.~\eqref{gnc:skew}.

\subsection{\texorpdfstring{$\mathfrak{so}(n)$}{so(n)} is the skew-symmetric matrices}
Take $G=SO(n)=\{g\in GL(n,\mathbb R):g^Tg=I,\ \det g=1\}$ and a curve
$g(t)$ in $SO(n)$ with $g(0)=I$ and $X=\frac{dg}{dt}(0)$. Differentiate
$g(t)^Tg(t)=I$ by the product rule and evaluate at $t=0$:
\begin{align}
 \frac{dg}{dt}(t)^T g(t)+g(t)^T\frac{dg}{dt}(t)&=0,\notag\\
 X^T+X&=0 .\label{quat:skew}
\end{align}
Conversely, let $X^T=-X$. Then $X^T$ and $X$ commute, so
$\exp(tX)^T\exp(tX)=\exp(tX^T)\exp(tX)=\exp(-tX)\exp(tX)=I$, and
$\det\exp(tX)=e^{t\operatorname{tr}X}=e^0=1$ because the diagonal of a
skew-symmetric matrix vanishes. By Eq.~\eqref{quat:matrix-lie-algebra},
\begin{equation}\label{quat:so-n}
 \mathfrak{so}(n)=\{X\in\mathfrak{gl}(n,\mathbb R):X^T=-X\},\qquad
 \dim\mathfrak{so}(n)=\frac{n(n-1)}{2},
\end{equation}
the dimension being the number of entries above the diagonal.

The regular-value argument in
\S\ref{quat:sec:orthogonal-submanifold} justifies the group dimension
in Eq.~\eqref{quat:dim-On}; Eq.~\eqref{quat:so-n} independently gives
the same dimension for its Lie algebra. Axis and angle describe rotations,
but require identifications at zero angle and at angle $\pi$ and are not
a single global coordinate chart.

\subsection{The basis \texorpdfstring{$E_{ab}$}{E\_ab} and its brackets}
Let $e_a$ be the standard basis column vectors of $\mathbb R^n$ and define
\begin{equation}\label{quat:Eab}
 E_{ab}=e_ae_b^T-e_be_a^T,\qquad
 (E_{ab})_{cd}=\delta_{ac}\delta_{bd}-\delta_{bc}\delta_{ad}.
\end{equation}
Then $E_{ba}=-E_{ab}$, and $\{E_{ab}:a<b\}$ is a basis of
$\mathfrak{so}(n)$. Using $e_b^Te_c=\delta_{bc}$, multiply out:
\begin{align}
 E_{ab}E_{cd}&=\delta_{bc}\,e_ae_d^T-\delta_{bd}\,e_ae_c^T
              -\delta_{ac}\,e_be_d^T+\delta_{ad}\,e_be_c^T,\notag\\
 E_{cd}E_{ab}&=\delta_{ad}\,e_ce_b^T-\delta_{bd}\,e_ce_a^T
              -\delta_{ac}\,e_de_b^T+\delta_{bc}\,e_de_a^T .\notag
\end{align}
Subtract the second line from the first and pair terms with the same
Kronecker delta:
\begin{align}
 [E_{ab},E_{cd}]
 &=\delta_{bc}\bigl(e_ae_d^T-e_de_a^T\bigr)
  -\delta_{bd}\bigl(e_ae_c^T-e_ce_a^T\bigr)\notag\\
 &\quad-\delta_{ac}\bigl(e_be_d^T-e_de_b^T\bigr)
  +\delta_{ad}\bigl(e_be_c^T-e_ce_b^T\bigr)\notag\\
 &=\delta_{bc}E_{ad}-\delta_{bd}E_{ac}-\delta_{ac}E_{bd}+\delta_{ad}E_{bc}.
 \label{quat:so-n-brackets}
\end{align}
Equation~\eqref{quat:so-n-brackets} holds for every $n$. A pair of
generators has a nonzero bracket only when the two planes share an axis.

\subsection{Three dimensions: Frankel's \texorpdfstring{$E_i$}{E\_i}}
For $n=3$ the dimension is $3$, and each coordinate plane has a unique
normal axis. Define
\begin{equation}\label{quat:Ei-from-Eab}
 E_i=-\tfrac12\epsilon_{ijk}E_{jk}.
\end{equation}
By Eq.~\eqref{quat:Eab},
\begin{equation}\label{quat:Ei-entries}
 (E_i)_{ab}=-\tfrac12\epsilon_{ijk}
   \bigl(\delta_{ja}\delta_{kb}-\delta_{ka}\delta_{jb}\bigr)
 =-\tfrac12\bigl(\epsilon_{iab}-\epsilon_{iba}\bigr)
 =-\epsilon_{iab}.
\end{equation}
Reading off the entries, $(E_1)_{23}=-1$, $(E_1)_{32}=1$, and similarly
for $E_2$ and $E_3$:
\begin{equation}\label{quat:frankel-Ei}
 E_1=\begin{bmatrix}0&0&0\\0&0&-1\\0&1&0\end{bmatrix},\qquad
 E_2=\begin{bmatrix}0&0&1\\0&0&0\\-1&0&0\end{bmatrix},\qquad
 E_3=\begin{bmatrix}0&-1&0\\1&0&0\\0&0&0\end{bmatrix}.
\end{equation}
These are Frankel's Eq.~(19.1), p.~492, and the basis
$(E_1,E_2,E_3)$ recorded from Kambe in \S\ref{dgdt:sec:kambe-so3}.
In the notation of Eq.~\eqref{quat:Eab}, $E_1=E_{32}$, $E_2=E_{13}$,
$E_3=E_{21}$.

Frankel obtains $E_3$ from the rotations about the $z$ axis,
\begin{equation}\label{quat:Rz}
 R(\theta)=\begin{bmatrix}\cos\theta&-\sin\theta&0\\
 \sin\theta&\cos\theta&0\\0&0&1\end{bmatrix}.
\end{equation}
The addition formulas give $R(\theta)R(\varphi)=R(\theta+\varphi)$: the
upper-left block of the product is
\begin{equation}
 \begin{bmatrix}\cos\theta\cos\varphi-\sin\theta\sin\varphi&
  -\cos\theta\sin\varphi-\sin\theta\cos\varphi\\
  \sin\theta\cos\varphi+\cos\theta\sin\varphi&
  -\sin\theta\sin\varphi+\cos\theta\cos\varphi\end{bmatrix}
 =\begin{bmatrix}\cos(\theta+\varphi)&-\sin(\theta+\varphi)\\
  \sin(\theta+\varphi)&\cos(\theta+\varphi)\end{bmatrix},
\end{equation}
and the third row and column are those of $I$. So $R$ is a one-parameter
subgroup, and its generator is its derivative at the identity. Differentiate
entry by entry, using $\frac{d}{d\theta}\cos\theta=-\sin\theta$ and
$\frac{d}{d\theta}\sin\theta=\cos\theta$, then set $\theta=0$:
\begin{equation}\label{quat:E3-derivative}
 \frac{dR}{d\theta}(0)
 =\begin{bmatrix}-\sin0&-\cos0&0\\\cos0&-\sin0&0\\0&0&0\end{bmatrix}
 =\begin{bmatrix}0&-1&0\\1&0&0\\0&0&0\end{bmatrix}=E_3 .
\end{equation}
In the other direction, sum $\exp(\theta E_3)$. Let
$P=\operatorname{diag}(1,1,0)$. Direct multiplication gives $E_3^2=-P$, and
$P^2=P$, $PE_3=E_3$. Hence for $m\ge1$
\begin{equation}
 E_3^{2m}=(-1)^mP,\qquad E_3^{2m+1}=(-1)^mE_3 ,
\end{equation}
the second also for $m=0$. Separate the $m=0$ even term and split the
series into even and odd powers:
\begin{align}
 \exp(\theta E_3)
 &=I+\sum_{m=1}^{\infty}\frac{(-1)^m\theta^{2m}}{(2m)!}\,P
   +\sum_{m=0}^{\infty}\frac{(-1)^m\theta^{2m+1}}{(2m+1)!}\,E_3\notag\\
 &=I+(\cos\theta-1)P+\sin\theta\,E_3 .\label{quat:exp-E3}
\end{align}
The first sum is the cosine series without its leading $1$. The diagonal
of Eq.~\eqref{quat:exp-E3} is $(\cos\theta,\cos\theta,1)$ and its
off-diagonal part is $\sin\theta\,E_3$, which is Eq.~\eqref{quat:Rz}.
Thus $R(\theta)=\exp(\theta E_3)$.

\subsection{The brackets of \texorpdfstring{$\mathfrak{so}(3)$}{so(3)}}
Frankel states $[E_1,E_2]=E_3$. Multiply the matrices of
Eq.~\eqref{quat:frankel-Ei}. Row $2$ of $E_1$ is $-e_3^T$, so row $2$ of
$E_1E_2$ is minus row $3$ of $E_2$; row $3$ of $E_1$ is $e_2^T$, so row $3$
of $E_1E_2$ is row $2$ of $E_2$, which is zero. Likewise for $E_2E_1$:
\begin{equation}
 E_1E_2=\begin{bmatrix}0&0&0\\1&0&0\\0&0&0\end{bmatrix},\qquad
 E_2E_1=\begin{bmatrix}0&1&0\\0&0&0\\0&0&0\end{bmatrix},\qquad
 E_1E_2-E_2E_1=\begin{bmatrix}0&-1&0\\1&0&0\\0&0&0\end{bmatrix}=E_3 .
\end{equation}
All nine brackets follow at once from the action of $E_i$ on vectors. By
Eq.~\eqref{quat:Ei-entries}, for $\mathbf r\in\mathbb R^3$,
\begin{equation}\label{quat:Ei-cross}
 (E_i\mathbf r)_a=-\epsilon_{iab}r^b=\epsilon_{aib}r^b
 =(\mathbf e_i\times\mathbf r)_a ,
\end{equation}
using $(\mathbf u\times\mathbf v)_a=\epsilon_{acb}u^cv^b$ with
$\mathbf u=\mathbf e_i$. The Jacobi identity for the cross product,
$\mathbf a\times(\mathbf b\times\mathbf r)+\mathbf b\times(\mathbf r\times\mathbf a)
+\mathbf r\times(\mathbf a\times\mathbf b)=0$, rearranges to
$\mathbf a\times(\mathbf b\times\mathbf r)-\mathbf b\times(\mathbf a\times\mathbf r)
=(\mathbf a\times\mathbf b)\times\mathbf r$. With
$\mathbf a=\mathbf e_i$, $\mathbf b=\mathbf e_j$,
\begin{align}
 [E_i,E_j]\mathbf r
 &=\mathbf e_i\times(\mathbf e_j\times\mathbf r)
  -\mathbf e_j\times(\mathbf e_i\times\mathbf r)\notag\\
 &=(\mathbf e_i\times\mathbf e_j)\times\mathbf r
  =\epsilon_{ijk}\,\mathbf e_k\times\mathbf r
  =\epsilon_{ijk}E_k\mathbf r .\notag
\end{align}
Since $\mathbf r$ is arbitrary,
\begin{equation}\label{quat:so3-brackets}
 [E_i,E_j]=\epsilon_{ijk}E_k ,\qquad c_{ij}{}^k=\epsilon_{ijk},
\end{equation}
Frankel's Eq.~(19.2) and his structure constants of $SO(3)$.

\subsection{Rotation about an axis and the angular velocity vector}
Frankel's Figure~19.1 (p.~492) shows a point $\mathbf r$ carried around
the $z$ axis, and he states $\frac{d\mathbf r}{dt}(0)=\boldsymbol\omega
\times\mathbf r(0)$. This follows from what a rotation about an axis does
to a vector, together with the definition of the angular velocity vector.

Let $\mathbf n$ be a unit vector and $\mathbf r_0\in\mathbb R^3$. Split
$\mathbf r_0$ into its parts along and perpendicular to $\mathbf n$:
\begin{equation}\label{quat:split}
 \mathbf r_0=(\mathbf n\cdot\mathbf r_0)\,\mathbf n+\mathbf r_\perp ,
 \qquad \mathbf r_\perp=\mathbf r_0-(\mathbf n\cdot\mathbf r_0)\,\mathbf n,
 \qquad \mathbf n\cdot\mathbf r_\perp=0 .
\end{equation}
Since $\mathbf n\times\mathbf n=0$, we have
$\mathbf n\times\mathbf r_0=\mathbf n\times\mathbf r_\perp$. This vector is
perpendicular to both $\mathbf n$ and $\mathbf r_\perp$, and has length
$|\mathbf n||\mathbf r_\perp|\sin\frac{\pi}{2}=|\mathbf r_\perp|$. So
$\mathbf r_\perp$ and $\mathbf n\times\mathbf r_0$ are perpendicular
vectors of equal length spanning the plane perpendicular to $\mathbf n$.
A rotation by the angle $\theta$ about $\mathbf n$, counterclockwise when
viewed from the tip of $\mathbf n$, leaves the component along $\mathbf n$
unchanged and turns $\mathbf r_\perp$ through $\theta$ toward
$\mathbf n\times\mathbf r_0$:
\begin{equation}\label{quat:rotate-about-n}
 \mathbf r(\theta)=(\mathbf n\cdot\mathbf r_0)\,\mathbf n
 +\cos\theta\,\mathbf r_\perp+\sin\theta\,\mathbf n\times\mathbf r_0 .
\end{equation}
For $\mathbf n=\mathbf e_3$ and $\mathbf r_0=(x,y,z)$, the perpendicular
part is $\mathbf r_\perp=(x,y,0)$ and $\mathbf e_3\times\mathbf r_0=(-y,x,0)$.
Equation~\eqref{quat:rotate-about-n} then gives
$(x\cos\theta-y\sin\theta,\;x\sin\theta+y\cos\theta,\;z)$, which is
$R(\theta)\mathbf r_0$ for the matrix of Eq.~\eqref{quat:Rz}. So
``counterclockwise about $\mathbf n$'' is the same sense as Frankel's
$R(\theta)$.

Now let the angle change in time, $\theta=\theta(t)$ with $\theta(0)=0$,
about the fixed axis $\mathbf n$. The \emph{angular velocity vector} is
defined as
\begin{equation}\label{quat:omega-def}
 \boldsymbol\omega=\frac{d\theta}{dt}\,\mathbf n ,
\end{equation}
the rate of turning times the axis, oriented by the right-hand rule.
Differentiate Eq.~\eqref{quat:rotate-about-n} with $\theta=\theta(t)$ by the
chain rule. The first term is constant:
\begin{equation}
 \frac{d\mathbf r}{dt}
 =\frac{d\theta}{dt}\Bigl(-\sin\theta\,\mathbf r_\perp
   +\cos\theta\,\mathbf n\times\mathbf r_0\Bigr).
\end{equation}
At $t=0$, $\theta=0$, so $\sin\theta=0$ and $\cos\theta=1$:
\begin{equation}\label{quat:drdt-omega}
 \frac{d\mathbf r}{dt}(0)=\frac{d\theta}{dt}(0)\,\mathbf n\times\mathbf r_0
 =\boldsymbol\omega\times\mathbf r(0).
\end{equation}
This is Frankel's statement. It is the definition~\eqref{quat:omega-def}
applied to the geometry of Eq.~\eqref{quat:rotate-about-n}, not a separate
fact.

\subsection{One-parameter subgroups are exponentials}
Frankel next says that the rotations at constant angular velocity form a
one-parameter subgroup, ``of the form $R(t)=e^{tS}$ for some skew symmetric
matrix $S$''. Two facts are used: these rotations form a one-parameter
subgroup, and every one-parameter subgroup is an exponential.

\emph{The rotations about a fixed axis form a one-parameter subgroup.} Let
$R_{\mathbf n}(\theta)$ denote the linear map~\eqref{quat:rotate-about-n}.
Choose a unit vector $\mathbf u\perp\mathbf n$ and put
$\mathbf v=\mathbf n\times\mathbf u$. Then $(\mathbf u,\mathbf v,\mathbf n)$
is a right-handed orthonormal basis, since
$\mathbf u\times\mathbf v=\mathbf u\times(\mathbf n\times\mathbf u)
=(\mathbf u\cdot\mathbf u)\mathbf n-(\mathbf u\cdot\mathbf n)\mathbf u
=\mathbf n$. Applying Eq.~\eqref{quat:rotate-about-n} to $\mathbf u$,
$\mathbf v$ and $\mathbf n$, and using
$\mathbf n\times\mathbf v=\mathbf n\times(\mathbf n\times\mathbf u)=-\mathbf u$,
\begin{equation}
 R_{\mathbf n}(\theta)\mathbf u=\cos\theta\,\mathbf u+\sin\theta\,\mathbf v,
 \quad
 R_{\mathbf n}(\theta)\mathbf v=-\sin\theta\,\mathbf u+\cos\theta\,\mathbf v,
 \quad
 R_{\mathbf n}(\theta)\mathbf n=\mathbf n .
\end{equation}
In the basis $(\mathbf u,\mathbf v,\mathbf n)$ the matrix of
$R_{\mathbf n}(\theta)$ is therefore the matrix of Eq.~\eqref{quat:Rz}, and
the addition formulas shown there give
$R_{\mathbf n}(\theta)R_{\mathbf n}(\varphi)=R_{\mathbf n}(\theta+\varphi)$.
At constant angular velocity, $\theta(t)=\frac{d\theta}{dt}t$, so
$R(t)=R_{\mathbf n}(\frac{d\theta}{dt}t)$ satisfies
$R(t)R(s)=R(t+s)$ and $R(0)=I$.

\emph{Every differentiable one-parameter subgroup is an exponential.} Let
$R\colon\mathbb R\to GL(n,\mathbb R)$ be differentiable with $R(0)=I$ and
$R(t+s)=R(t)R(s)$ for all $s,t$, and put $S=\frac{dR}{dt}(0)$. Differentiate
$R(t+s)=R(t)R(s)$ with respect to $s$, holding $t$ fixed, and set $s=0$:
\begin{equation}\label{quat:one-param-ode}
 \frac{dR}{dt}(t)=R(t)S .
\end{equation}
Let $M(t)=R(t)\exp(-tS)$. Since
$\frac{d}{dt}\exp(-tS)=-S\exp(-tS)$, the product rule and
Eq.~\eqref{quat:one-param-ode} give
\begin{equation}
 \frac{dM}{dt}(t)=R(t)S\exp(-tS)-R(t)S\exp(-tS)=0 .
\end{equation}
So $M(t)=M(0)=I$, and multiplying on the right by $\exp(tS)$,
\begin{equation}\label{quat:one-param-exp}
 R(t)=\exp(tS),\qquad S=\frac{dR}{dt}(0).
\end{equation}
This is the content of Theorem~\ref{dgdt:Thm:5.1YKos}(ii) for one-parameter
subgroups, and of Frankel's Section~15.2 (p.~398). When $R(t)\in SO(3)$, the
generator $S$ is skew-symmetric by Eq.~\eqref{quat:skew}.

\emph{Identifying $S$.} With $\mathbf r(t)=R(t)\mathbf r_0
=\exp(tS)\mathbf r_0$, differentiation at $t=0$ gives
$\frac{d\mathbf r}{dt}(0)=S\mathbf r_0$. Comparing with
Eq.~\eqref{quat:drdt-omega},
\begin{equation}\label{quat:S-omega}
 S\mathbf r_0=\boldsymbol\omega\times\mathbf r_0
 \quad\text{for every }\mathbf r_0\in\mathbb R^3 .
\end{equation}

\subsection{The generators act by cross products}
Frankel notes that $E_j(\mathbf r)=\mathbf e_j\times\mathbf r$. Check it
for each $j$ with $\mathbf r=(x,y,z)$. Multiplying the matrices of
Eq.~\eqref{quat:frankel-Ei} into $\mathbf r$, and computing the cross
products from $\mathbf a\times\mathbf b=(a^2b^3-a^3b^2,\;a^3b^1-a^1b^3,\;
a^1b^2-a^2b^1)$,
\begin{align}
 E_1\mathbf r&=(0,\,-z,\,y),&
 \mathbf e_1\times\mathbf r&=(0\cdot z-0\cdot y,\;0\cdot x-1\cdot z,\;
   1\cdot y-0\cdot x)=(0,\,-z,\,y),\notag\\
 E_2\mathbf r&=(z,\,0,\,-x),&
 \mathbf e_2\times\mathbf r&=(1\cdot z-0\cdot y,\;0\cdot x-0\cdot z,\;
   0\cdot y-1\cdot x)=(z,\,0,\,-x),\notag\\
 E_3\mathbf r&=(-y,\,x,\,0),&
 \mathbf e_3\times\mathbf r&=(0\cdot z-1\cdot y,\;1\cdot x-0\cdot z,\;
   0\cdot y-0\cdot x)=(-y,\,x,\,0).\label{quat:Ej-explicit}
\end{align}
In index form this is Eq.~\eqref{quat:Ei-cross}, used above for the
brackets. The cross product is
linear in its first factor, so for $\boldsymbol\omega=\omega^j\mathbf e_j$
\begin{equation}\label{quat:hat}
 \boldsymbol\omega\times\mathbf r=\omega^j\,\mathbf e_j\times\mathbf r
 =\omega^jE_j\mathbf r ,\qquad
 \widehat{\boldsymbol\omega}:=\omega^jE_j .
\end{equation}
This $\widehat{\boldsymbol\omega}$ is the hat map of the rigid-body part,
defined after Eq.~\eqref{gnc:skew}, and it is Frankel's
$\mathbf E\bullet\boldsymbol\omega$. Holm calls the same map the hat map.

Equations~\eqref{quat:S-omega} and \eqref{quat:hat} give
$S=\widehat{\boldsymbol\omega}$, and Eq.~\eqref{quat:one-param-exp} becomes
\begin{equation}\label{quat:frankel-19-3}
 R(t)=\exp\bigl(t\,\omega^jE_j\bigr)=\exp\bigl(t\,\widehat{\boldsymbol\omega}\bigr),
\end{equation}
Frankel's Eq.~(19.3), p.~493. At constant rate the angle is
$\theta=t\frac{d\theta}{dt}$, so $t\boldsymbol\omega=\theta\mathbf n$ by
Eq.~\eqref{quat:omega-def}, and
\begin{equation}\label{quat:frankel-19-4}
 R_{\mathbf n}(\theta)=\exp\bigl(\theta\,n^jE_j\bigr),
\end{equation}
Frankel's Eq.~(19.4): the rotation through $\theta$ about the unit vector
$\mathbf n$.

By Eq.~\eqref{quat:so3-brackets}, the hat map is a Lie algebra
isomorphism from $(\mathbb R^3,\times)$ to $\mathfrak{so}(3)$:
\begin{equation}
 [\widehat{\mathbf u},\widehat{\mathbf v}]=u^iv^j[E_i,E_j]
 =u^iv^j\epsilon_{ijk}E_k=(\mathbf u\times\mathbf v)^kE_k
 =\widehat{\mathbf u\times\mathbf v}.
\end{equation}

\section{Two bases of \texorpdfstring{$\mathfrak{su}(2)$}{su(2)}: Kosmann-Schwarzbach and Frankel}
\S\ref{dgdt:sec:KS-su2-bases} records Kosmann-Schwarzbach's basis
\begin{equation}\label{quat:KS-xi}
 \xi_1=\frac i2\begin{bmatrix}0&1\\1&0\end{bmatrix},\qquad
 \xi_2=\frac12\begin{bmatrix}0&-1\\1&0\end{bmatrix},\qquad
 \xi_3=\frac i2\begin{bmatrix}1&0\\0&-1\end{bmatrix},\qquad
 [\xi_k,\xi_l]=\epsilon_{klm}\xi_m .
\end{equation}
Frankel's basis is $F_a=\sigma_a/(2i)$. From his Eq.~(19.6),
$[\sigma_a,\sigma_b]=2i\epsilon_{abc}\sigma_c$,
\begin{equation}\label{quat:Fa-brackets}
 [F_a,F_b]=\frac{1}{(2i)^2}[\sigma_a,\sigma_b]
 =-\frac14\,2i\,\epsilon_{abc}\sigma_c
 =\epsilon_{abc}\frac{\sigma_c}{2i}
 =\epsilon_{abc}F_c ,
\end{equation}
using $-\frac{i}{2}=\frac{1}{2i}$. Both bases therefore have the structure
constants of Eq.~\eqref{quat:so3-brackets}, and the linear maps
$\xi_k\mapsto E_k$ and $F_a\mapsto E_a$ are both Lie algebra isomorphisms
$\mathfrak{su}(2)\to\mathfrak{so}(3)$. The second is Frankel's
Eq.~(19.14), $\mathrm{Ad}_*(\sigma_a/2i)=E_a$.

The two isomorphisms differ. Compare entries with the Pauli matrices of
Eq.~\eqref{quat:pauli}: $\xi_1=\frac i2\sigma_1$, while
$F_1=\frac{1}{2i}\sigma_1=-\frac i2\sigma_1$; next
$F_2=\frac{1}{2i}\begin{bmatrix}0&-i\\i&0\end{bmatrix}
=\frac12\begin{bmatrix}0&-1\\1&0\end{bmatrix}$; and
$\xi_3=\frac i2\sigma_3=-F_3$. Thus
\begin{equation}\label{quat:KS-vs-Frankel}
 (\xi_1,\xi_2,\xi_3)=(-F_1,\,F_2,\,-F_3).
\end{equation}
This change of basis is conjugation by an element of $SU(2)$. Let
$u=-i\sigma_2=\begin{bmatrix}0&-1\\1&0\end{bmatrix}$, so $u^{-1}=i\sigma_2$.
Since distinct Pauli matrices anticommute (Frankel's Eq.~(19.18)) and
$\sigma_2^2=I$,
\begin{align}
 u\sigma_1u^{-1}&=\sigma_2\sigma_1\sigma_2=-\sigma_1\sigma_2\sigma_2=-\sigma_1,\notag\\
 u\sigma_2u^{-1}&=\sigma_2\sigma_2\sigma_2=\sigma_2,\notag\\
 u\sigma_3u^{-1}&=\sigma_2\sigma_3\sigma_2=-\sigma_3\sigma_2\sigma_2=-\sigma_3 ,
 \label{quat:conjugate-by-j}
\end{align}
where the factors $(-i)(i)=1$ cancel. Dividing by $2i$ and comparing with
Eq.~\eqref{quat:KS-vs-Frankel},
$\xi_a=uF_au^{-1}=\mathrm{Ad}(u)F_a$. Under
Eq.~\eqref{quat:dictionary} $u$ is the quaternion $e_2=j$, which is
Eq.~\eqref{quat:axis-angle} with $\theta=\pi$ and $\mathbf A=\mathbf e_2$:
the rotation by $\pi$ about the second axis,
$\operatorname{diag}(-1,1,-1)$, as the signs in
Eq.~\eqref{quat:KS-vs-Frankel} show.

This is an instance of a general fact. Two bases with the same structure
constants determine the same Lie algebra, but the isomorphism between them
is fixed only up to an automorphism. For $\mathfrak{so}(3)$, identified
with $(\mathbb R^3,\times)$ by Eq.~\eqref{quat:hat}, the automorphisms are
exactly $SO(3)$. Let $A\in GL(3,\mathbb R)$ satisfy
$A(\mathbf u\times\mathbf v)=A\mathbf u\times A\mathbf v$ for all
$\mathbf u,\mathbf v$.

\emph{$A$ is orthogonal.} The Killing form $B(X,Y)=\operatorname{tr}
(\operatorname{ad}_X\operatorname{ad}_Y)$ is preserved by automorphisms,
since $A[X,A^{-1}Y]=[AX,Y]$ gives
$\operatorname{ad}_{AX}=A\operatorname{ad}_XA^{-1}$ and the trace is
invariant under conjugation. On $(\mathbb R^3,\times)$,
$\operatorname{ad}_{\mathbf e_i}=E_i$ by Eq.~\eqref{quat:Ei-cross}, so by
Eq.~\eqref{quat:Ei-entries}
\begin{equation}
 B(\mathbf e_i,\mathbf e_j)=(E_i)_{ab}(E_j)_{ba}
 =\epsilon_{iab}\epsilon_{jba}=-\epsilon_{iab}\epsilon_{jab}=-2\delta_{ij}.
\end{equation}
Hence $A$ preserves the Euclidean inner product, $A^TA=I$.

\emph{$\det A=1$.} For any $A\in GL(3,\mathbb R)$ and all
$\mathbf u,\mathbf v,\mathbf w$,
\begin{equation}
 (A\mathbf u\times A\mathbf v)\cdot A\mathbf w
 =\det[A\mathbf u,A\mathbf v,A\mathbf w]
 =\det A\,\det[\mathbf u,\mathbf v,\mathbf w]
 =\det A\,(\mathbf u\times\mathbf v)\cdot\mathbf w ,
\end{equation}
so $A^T(A\mathbf u\times A\mathbf v)=\det A\,(\mathbf u\times\mathbf v)$.
With $A^T=A^{-1}$ and $A\mathbf u\times A\mathbf v=A(\mathbf u\times\mathbf v)$
this reads $\mathbf u\times\mathbf v=\det A\,(\mathbf u\times\mathbf v)$.
Choosing $\mathbf u\times\mathbf v\neq0$ gives $\det A=1$. Conversely, the
same identity with $A^T=A^{-1}$ and $\det A=1$ shows that every
$R\in SO(3)$ preserves the cross product. Through Eq.~\eqref{quat:hat} this
reads $R\,\widehat{\mathbf v}\,R^{-1}=\widehat{R\mathbf v}$, so each
automorphism of $\mathfrak{so}(3)$ is $\mathrm{Ad}(R)$ for a rotation $R$.
Kosmann-Schwarzbach's and Frankel's identifications of $\mathfrak{su}(2)$
with $\mathfrak{so}(3)$ differ by the rotation
$\operatorname{diag}(-1,1,-1)=\mathrm{Ad}(j)$.

\subsection{Kosmann-Schwarzbach's covering map \texorpdfstring{$\varphi$}{phi}}\label{quat:sec:KS-phi}
Kosmann-Schwarzbach builds her covering map from the same sign choice. Her
quaternion basis of $\mathfrak{su}(2)$ is
\cite[Ch.~5 \S1.1(d), p.~72]{dgdt:YKosmann-Schwarzbach2010}
\begin{equation}\label{quat:KS-IJK}
 \mathcal I=2\xi_1=i\sigma_1,\qquad
 \mathcal J=2\xi_2=-i\sigma_2,\qquad
 \mathcal K=2\xi_3=i\sigma_3,
\end{equation}
with $\mathcal I^2=\mathcal J^2=\mathcal K^2=-I$, $\mathcal I\mathcal J=\mathcal K$,
$\mathcal J\mathcal K=\mathcal I$, $\mathcal K\mathcal I=\mathcal J$, and she
identifies the quaternion units $i,j,k$ with $\mathcal I,\mathcal J,\mathcal K$
(p.~73). Under Eq.~\eqref{quat:dictionary} these are
$(-e_1,e_2,-e_3)=(je_1j^{-1},\,je_2j^{-1},\,je_3j^{-1})$; for instance
$je_1j^{-1}=-je_1j=-(-e_3)j=e_3e_2=-e_1$. So her units are ours conjugated
by $j$, again Hamilton's order since conjugation is an algebra
automorphism.

She identifies $\mathbb R^3$ with $\mathfrak{su}(2)$ by
$a\mapsto X_a=a^1\mathcal I+a^2\mathcal J+a^3\mathcal K$ (\S2.2, p.~77), and
$\det X_a=\|a\|^2$ (her Eq.~(1.1), p.~73). Put
$D=\operatorname{diag}(-1,1,-1)$. By Eq.~\eqref{quat:KS-IJK},
\begin{equation}\label{quat:KS-Xa}
 X_a=i a^1\sigma_1-i a^2\sigma_2+i a^3\sigma_3
 =-i\bigl((-a^1)\sigma_1+a^2\sigma_2+(-a^3)\sigma_3\bigr)
 =-i\Sigma(Da).
\end{equation}
Her map $\varphi\colon SU(2)\to GL(3,\mathbb R)$ sends $g$ to the matrix of
$\mathrm{Ad}_g$ in the basis $(\mathcal I,\mathcal J,\mathcal K)$
(\S2.3, pp.~77--78), that is, $\mathrm{Ad}_gX_a=X_{\varphi(g)a}$. Compute the
left side with Eq.~\eqref{quat:Phi}, $g\Sigma(\mathbf x)g^{-1}=\Sigma(\Phi(g)\mathbf x)$,
and $D^2=I$:
\begin{equation}
 \mathrm{Ad}_gX_a=-i\,g\Sigma(Da)g^{-1}=-i\Sigma\bigl(\Phi(g)Da\bigr)
 =-i\Sigma\bigl(D\,(D\Phi(g)D)\,a\bigr)=X_{D\Phi(g)Da}.
\end{equation}
Since $a\mapsto X_a$ is a bijection, $\varphi(g)=D\Phi(g)D^{-1}$. By
Eq.~\eqref{quat:conjugate-by-j}, $u=-i\sigma_2$ satisfies
$u\Sigma(\mathbf e_1)u^{-1}=\Sigma(-\mathbf e_1)$,
$u\Sigma(\mathbf e_2)u^{-1}=\Sigma(\mathbf e_2)$,
$u\Sigma(\mathbf e_3)u^{-1}=\Sigma(-\mathbf e_3)$, so $\Phi(u)=D$. Because
$\Phi$ is a homomorphism,
\begin{equation}\label{quat:KS-phi}
 \varphi(g)=\Phi(u)\Phi(g)\Phi(u)^{-1}=\Phi(ugu^{-1}),\qquad
 \varphi=\Phi\circ\mathcal C_u,\qquad u=-i\sigma_2 .
\end{equation}
Thus $\varphi$ is $\Phi$ preceded by an automorphism of $SU(2)$, and
everything proved for $\Phi$ transfers. $\varphi$ is onto $SO(3)$ because
$\mathcal C_u$ is onto $SU(2)$. Also $\varphi(g)=I$ if and only if
$ugu^{-1}\in\ker\Phi=\{\pm I\}$, if and only if $g=\pm I$, which is her
$\operatorname{Ker}\varphi=\{I,-I\}$ (p.~78). She concludes that $SU(2)$
is the double-sheeted universal cover of $SO(3)$ (Proposition~2.6, p.~78).

\emph{Her half angle.} Kosmann-Schwarzbach writes $\mathrm{Rot}(a,t)$ for
the rotation through $t$ about the unit vector $a$,
$\mathrm{Rot}(a,t)x=x+(1-\cos t)\,a\wedge(a\wedge x)+\sin t\,a\wedge x$
(her Eq.~(2.1), p.~74). This is our $R_{\mathbf n}(\theta)$: expanding
$a\wedge(a\wedge x)=(a\cdot x)a-x$ and writing $x=(a\cdot x)a+x_\perp$
gives Eq.~\eqref{quat:rotate-about-n}.
Her Proposition~2.5(iii) (p.~78) states
\begin{equation}\label{quat:KS-half-angle}
 \varphi\bigl(\exp(tX_a)\bigr)=\mathrm{Rot}(a,2t)\qquad(\|a\|=1).
\end{equation}
Check it against Eq.~\eqref{quat:two-preimages}. By her Proposition~2.4
(p.~77), $\exp(tX)=\cos t\,I+\sin t\,X$ when $X\in\mathfrak{su}(2)$ and
$\det X=1$. With Eq.~\eqref{quat:KS-Xa},
$\exp(tX_a)=\cos t\,I-i\sin t\,\Sigma(Da)$, which is
Eq.~\eqref{quat:two-preimages} with $\theta=2t$ and $\mathbf n=Da$. So
$\Phi(\exp(tX_a))=R_{Da}(2t)$, and by Eq.~\eqref{quat:KS-phi}
\begin{equation}
 \varphi\bigl(\exp(tX_a)\bigr)=D\,R_{Da}(2t)\,D^{-1}=R_{D^2a}(2t)=R_a(2t),
\end{equation}
using $gR_{\mathbf n}(\theta)g^{-1}=R_{g\mathbf n}(\theta)$ for
$g\in SO(3)$ (her Eq.~(2.2), p.~75; $g$ preserves the cross product, so
$gR_{\mathbf n}(\theta)g^{-1}$ fixes $g\mathbf n$ and turns the plane
perpendicular to it through $\theta$ in the same sense). Her parameter $t$
is our half angle $\theta/2$.

\emph{Why her signs.} Her Proposition~2.5(i) (p.~78) says the matrix of
$\mathrm{ad}_{\xi_k}$ in the basis $(\xi_1,\xi_2,\xi_3)$ is $\eta_k$, and
her $\eta_k$ (\S1.2, p.~73) are the matrices $E_k$ of
Eq.~\eqref{quat:Ei-entries}. So $d\varphi(\xi_k)=E_k$ with no signs, and
her Lie algebra isomorphism $\xi_k\mapsto\eta_k$ (p.~73) is exactly
$d\varphi$. For $\Phi$ the same basis gives
$d\Phi(\xi_k)=(-E_1,E_2,-E_3)$ (\S\ref{quat:sec:covering}); differentiating
$\varphi=D\Phi D^{-1}$ gives $d\varphi(\xi_k)=D\,d\Phi(\xi_k)D^{-1}$, and
$DE_1D^{-1}=\widehat{D\mathbf e_1}=-E_1$, $DE_2D^{-1}=E_2$,
$DE_3D^{-1}=-E_3$ by $R\widehat{\mathbf v}R^{-1}=\widehat{R\mathbf v}$,
which cancels the signs. Frankel's choice makes $F_a\mapsto E_a$ equal to
$d\Phi$; Kosmann-Schwarzbach's makes $\xi_k\mapsto\eta_k$ equal to
$d\varphi$. Each author chose the $\mathbb R^3$ identification that makes
their own basis map to $E_k$.

\emph{Isomorphism theorem.} Kosmann-Schwarzbach states only that a kernel
is a normal subgroup and that $p\colon G\to G/H$ is a morphism (Ch.~1,
p.~2). The isomorphism $SU(2)/\{\pm I\}\cong SO(3)$ is
Theorem~\ref{quat:thm:first-iso} applied to $\varphi$ or $\Phi$.

\section{The quaternion basis in Frankel and in Holm}
Frankel \cite{TFrankel2012} works with the Pauli matrices
\begin{equation}\label{quat:pauli}
 \sigma_1=\begin{bmatrix}0&1\\1&0\end{bmatrix},\qquad
 \sigma_2=\begin{bmatrix}0&-i\\i&0\end{bmatrix},\qquad
 \sigma_3=\begin{bmatrix}1&0\\0&-1\end{bmatrix}
\end{equation}
(his Eq.~(19.5), p.~493). Here the $i$ inside $\sigma_2$ is the complex
unit, not the quaternion $e_1$. His product relations, Eq.~(19.17), p.~499,
are
\begin{equation}\label{quat:pauli-products}
 \sigma_1\sigma_2=i\sigma_3,\qquad
 \sigma_2\sigma_3=i\sigma_1,\qquad
 \sigma_3\sigma_1=i\sigma_2,\qquad
 \sigma_a^2=I .
\end{equation}
Holm \cite{DDHolm2011} writes a quaternion as a tetrad
$q=[q_0,\mathbf q]=q_0J_0+q^aJ_a$ and identifies $J_0=\sigma_0=I$ and
$J_a=-i\sigma_a$ (Section~3.1.1, Eqs.~(3.1.1)--(3.1.3), pp.~79--81).
Frankel uses $\sigma_a/i$, and $1/i=-i$, so both authors use the same
matrices:
\begin{equation}\label{quat:dictionary}
 e_a\;\longleftrightarrow\;J_a\;=\;-i\sigma_a\;=\;\frac{\sigma_a}{i}.
\end{equation}
Check that these matrices satisfy Eq.~\eqref{quat:basis}. Using
Eq.~\eqref{quat:pauli-products},
\begin{align}
 (-i\sigma_1)(-i\sigma_2)&=(-i)^2\sigma_1\sigma_2
   =-\sigma_1\sigma_2=-i\sigma_3,\notag\\
 (-i\sigma_a)^2&=(-i)^2\sigma_a^2=-I .\label{quat:dictionary-check}
\end{align}
The first line is $e_1e_2=e_3$ under Eq.~\eqref{quat:dictionary}; the
cyclic permutations of Eq.~\eqref{quat:pauli-products} give $e_2e_3=e_1$
and $e_3e_1=e_2$ in the same way. Both books therefore follow Hamilton's
order, and Eq.~\eqref{quat:dictionary} is an algebra isomorphism from
$\mathbb H$ onto the real span of $I$ and $-i\sigma_a$.

Two scalings differ from ours. Frankel's real basis of $\mathfrak{su}(2)$ is
$\sigma_a/(2i)=\tfrac12e_a$, which has the $\mathfrak{so}(3)$ brackets
(p.~493). His identification of $\mathbb R^3$ with traceless Hermitian
matrices, Eq.~(19.7), p.~494, is $\mathbf x_*=\Sigma(\mathbf x)=x^a\sigma_a$
(Eq.~\eqref{quat:Sigma}), while our pure quaternion is
\begin{equation}\label{quat:frankel-star}
 x^ae_a=x^a(-i\sigma_a)=-i\Sigma(\mathbf x),
\end{equation}
which is Holm's $\widetilde{\mathbf x}$.

\section{The half angle in both notations}
Frankel's Eq.~(19.15), p.~497, sends
$\exp\bigl(\tfrac{\boldsymbol\sigma}{2i}\cdot\mathbf A\theta\bigr)\in SU(2)$
to $\exp(\theta\,\mathbf E\cdot\mathbf A)\in SO(3)$, the rotation by
$\theta$ about the unit vector $\mathbf A$. To sum the left side, use his
anticommutator, Eq.~(19.18), p.~499:
$\sigma_a\sigma_b+\sigma_b\sigma_a=2\delta_{ab}I$. For a unit vector
$\mathbf A$,
\begin{align}
 (\boldsymbol\sigma\cdot\mathbf A)^2
 &=A^aA^b\sigma_a\sigma_b
  =\tfrac12A^aA^b\bigl(\sigma_a\sigma_b+\sigma_b\sigma_a\bigr)\notag\\
 &=A^aA^b\delta_{ab}I=I .\label{quat:sigma-square}
\end{align}
The symmetrization in the first line is allowed because $A^aA^b$ is
symmetric in $a,b$. Put $N=\boldsymbol\sigma\cdot\mathbf A/i
=A^a(-i\sigma_a)$, the pure unit quaternion $A^ae_a$ under
Eq.~\eqref{quat:dictionary}. Then Eq.~\eqref{quat:sigma-square} gives
$N^2=(1/i)^2I=-I$, hence
\begin{equation}
 N^{2m}=(-1)^mI,\qquad N^{2m+1}=(-1)^mN .
\end{equation}
Split the exponential series into even and odd powers:
\begin{align}
 \exp\Bigl(\frac{\theta}{2}N\Bigr)
 &=\sum_{m=0}^{\infty}\frac{(\theta/2)^{2m}}{(2m)!}N^{2m}
  +\sum_{m=0}^{\infty}\frac{(\theta/2)^{2m+1}}{(2m+1)!}N^{2m+1}\notag\\
 &=\sum_{m=0}^{\infty}\frac{(-1)^m(\theta/2)^{2m}}{(2m)!}\,I
  +\sum_{m=0}^{\infty}\frac{(-1)^m(\theta/2)^{2m+1}}{(2m+1)!}\,N\notag\\
 &=\cos\frac{\theta}{2}\,I+\sin\frac{\theta}{2}\,N .\label{quat:half-angle}
\end{align}
Since $N=-i\,\boldsymbol\sigma\cdot\mathbf A$, Eq.~\eqref{quat:half-angle}
is Frankel's $\cos(\theta/2)I-i\sin(\theta/2)\,\boldsymbol\sigma\cdot\mathbf A$
(p.~500). In the notation of Eq.~\eqref{quat:scalar-first} it is
\begin{equation}\label{quat:axis-angle}
 q=\cos\frac{\theta}{2}+\sin\frac{\theta}{2}\,A^ae_a .
\end{equation}
Replacing $\theta$ by $\theta+2\pi$ changes the signs of both
$\cos(\theta/2)$ and $\sin(\theta/2)$, so $q\mapsto-q$ while the rotation
$\exp(\theta\,\mathbf E\cdot\mathbf A)$ is unchanged.

\section{Reading quaternions geometrically}
Frankel \cite{TFrankel2012} is the spine. Printed pages refer to the third
edition. Read Sections~19.1--19.2 (pp.~491--503): $SO(3)$ and
$\mathfrak{so}(3)$, the Lie algebra $\mathfrak{su}(2)$, $SU(2)\cong S^3$,

Review Section~15.3 (pp.~402--406) for the exponential map when needed.
Then read Section~21.2 (pp.~567--575) on the fundamental group, covering

$\pi_1(SO(3))=\mathbb Z_2$ and $SU(2)$ as the universal cover.
Read the Hopf bundle in Section~17.4c (p.~473) and the Hopf map in
Section~22.4c (pp.~606--607), after the homotopy groups of
Section~22.3 (pp.~596--604). Read the spinor bundle and spin connection of
Section~19.5 (pp.~515--522) after the connections part of these notes.

Holm \cite{DDHolm2011} then puts the same group into mechanics.
Chapter~3 (pp.~77--110) covers quaternion multiplication through Pauli
matrices, Cayley--Klein parameters, pure quaternions and $SU(2)$, the tilde
