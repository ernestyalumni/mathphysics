\section{Lie Groups}


\subsection*{Basic Definitions}

\textbf{Lie group} smooth manifold $G$ s.t. multiplication map $\begin{aligned} & \quad \\ 
  & m : G \times G \to G \\
  & m(g,h) = gh \end{aligned}$ \\
inversion map $\begin{aligned} & \quad \\ 
  & i : G \to G \\
  & i(g) = g^{-1} \end{aligned}$ 
smooth.

\begin{proposition}[7.1]
If $(g,h) \mapsto gh^{-1}$ smooth, $G$ Lie group
\end{proposition}
\begin{exercise}[Lee 2e, Exercise 7.2, p.~151 \cite{dgdt:JLee2012}]
Prove Proposition 7.1.
\end{exercise}
\noindent\emph{Note.} Proposition~7.1: if $G$ is a smooth manifold with a group structure such that the map $G\times G\to G$ given by $(g,h)\mapsto gh^{-1}$ is smooth, then $G$ is a Lie group.
\paragraph{Solution.}
\begin{proof}
  $\forall \, g,h \in G$, $gh^2 \in G$ since $G$ group \\
$(gh^2,h) \mapsto gh$ smooth.  Define $m(g,h) = (gh^2,h) \mapsto gh$.  So $m$ smooth.  \\
$1\in G$ since $G$ group.  $(1,g) \mapsto g^{-1}$ smooth so $i(g) = g^{-1}$, defined this way, smooth.  
\end{proof}
